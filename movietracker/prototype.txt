import React, { useState, useMemo, useCallback } from 'react';
import { Plus, Edit3, Trophy, TrendingUp, Calendar, Target } from 'lucide-react';

const PowerRankingsSystem = () => {
  // Initialize with 14 teams
  const [teams, setTeams] = useState(() => {
    const teamNames = [
      'Team Alpha', 'Team Beta', 'Team Gamma', 'Team Delta', 'Team Echo',
      'Team Foxtrot', 'Team Golf', 'Team Hotel', 'Team India', 'Team Juliet',
      'Team Kilo', 'Team Lima', 'Team Mike', 'Team November'
    ];
    return teamNames.map((name, i) => ({
      id: i + 1,
      name,
      wins: Math.floor(Math.random() * 8),
      losses: Math.floor(Math.random() * 4),
      pointsFor: Math.floor(Math.random() * 200) + 150,
      pointsAgainst: Math.floor(Math.random() * 180) + 120,
      opponentWinPct: Math.random() * 0.4 + 0.3, // 30-70%
      qualityWins: Math.floor(Math.random() * 3),
      badLosses: Math.floor(Math.random() * 2),
      recentForm: Math.random() * 20 - 10 // -10 to +10
    }));
  });

  const [editingTeam, setEditingTeam] = useState(null);
  const [activeTab, setActiveTab] = useState('rankings');

  // Calculate enhanced metrics
  const enhancedTeams = useMemo(() => {
    const leagueAvgWinPct = teams.reduce((sum, team) => sum + (team.wins / (team.wins + team.losses)), 0) / teams.length;
    
    return teams.map(team => {
      const gamesPlayed = team.wins + team.losses;
      const winPct = gamesPlayed > 0 ? team.wins / gamesPlayed : 0;
      const pointDiff = team.pointsFor - team.pointsAgainst;
      const ppg = gamesPlayed > 0 ? team.pointsFor / gamesPlayed : 0;
      const papg = gamesPlayed > 0 ? team.pointsAgainst / gamesPlayed : 0;
      
      // Strength of Schedule
      const sos = team.opponentWinPct - leagueAvgWinPct;
      
      // Opponent-adjusted point differential
      const adjustedPointDiff = pointDiff + (sos * 50);
      
      // Base Rating
      const baseRating = (winPct * 40) + (adjustedPointDiff * 0.8) + (ppg * 0.2) - (papg * 0.2);
      
      // SOS Adjustment
      const sosAdjustment = (sos * 20) + (team.qualityWins * 3) - (team.badLosses * 2);
      
      // Final Power Rating
      const powerRating = (baseRating * 0.60) + (sosAdjustment * 0.25) + (team.recentForm * 0.15);
      
      return {
        ...team,
        gamesPlayed,
        winPct,
        pointDiff,
        ppg,
        papg,
        sos,
        adjustedPointDiff,
        baseRating,
        sosAdjustment,
        powerRating
      };
    }).sort((a, b) => b.powerRating - a.powerRating);
  }, [teams]);

  const updateTeam = useCallback((id, updates) => {
    setTeams(prev => prev.map(team => 
      team.id === id ? { ...team, ...updates } : team
    ));
  }, []);

  const TeamEditor = ({ team, onClose }) => (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50">
      <div className="bg-white rounded-xl p-6 w-full max-w-md">
        <h3 className="text-lg font-bold mb-4 flex items-center gap-2">
          <Edit3 size={20} />
          Edit {team.name}
        </h3>
        
        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium mb-1">Team Name</label>
            <input
              type="text"
              value={team.name}
              onChange={(e) => updateTeam(team.id, { name: e.target.value })}
              className="w-full p-2 border rounded-lg"
            />
          </div>
          
          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium mb-1">Wins</label>
              <input
                type="number"
                min="0"
                value={team.wins}
                onChange={(e) => updateTeam(team.id, { wins: parseInt(e.target.value) || 0 })}
                className="w-full p-2 border rounded-lg"
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">Losses</label>
              <input
                type="number"
                min="0"
                value={team.losses}
                onChange={(e) => updateTeam(team.id, { losses: parseInt(e.target.value) || 0 })}
                className="w-full p-2 border rounded-lg"
              />
            </div>
          </div>
          
          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium mb-1">Points For</label>
              <input
                type="number"
                min="0"
                value={team.pointsFor}
                onChange={(e) => updateTeam(team.id, { pointsFor: parseInt(e.target.value) || 0 })}
                className="w-full p-2 border rounded-lg"
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">Points Against</label>
              <input
                type="number"
                min="0"
                value={team.pointsAgainst}
                onChange={(e) => updateTeam(team.id, { pointsAgainst: parseInt(e.target.value) || 0 })}
                className="w-full p-2 border rounded-lg"
              />
            </div>
          </div>
          
          <div>
            <label className="block text-sm font-medium mb-1">Opponent Win %</label>
            <input
              type="number"
              min="0"
              max="1"
              step="0.001"
              value={team.opponentWinPct}
              onChange={(e) => updateTeam(team.id, { opponentWinPct: parseFloat(e.target.value) || 0 })}
              className="w-full p-2 border rounded-lg"
            />
          </div>
          
          <div className="grid grid-cols-3 gap-3">
            <div>
              <label className="block text-sm font-medium mb-1">Quality Wins</label>
              <input
                type="number"
                min="0"
                value={team.qualityWins}
                onChange={(e) => updateTeam(team.id, { qualityWins: parseInt(e.target.value) || 0 })}
                className="w-full p-2 border rounded-lg"
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">Bad Losses</label>
              <input
                type="number"
                min="0"
                value={team.badLosses}
                onChange={(e) => updateTeam(team.id, { badLosses: parseInt(e.target.value) || 0 })}
                className="w-full p-2 border rounded-lg"
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">Recent Form</label>
              <input
                type="number"
                min="-10"
                max="10"
                step="0.1"
                value={team.recentForm}
                onChange={(e) => updateTeam(team.id, { recentForm: parseFloat(e.target.value) || 0 })}
                className="w-full p-2 border rounded-lg"
              />
            </div>
          </div>
        </div>
        
        <div className="flex gap-2 mt-6">
          <button
            onClick={onClose}
            className="flex-1 px-4 py-2 bg-gray-200 text-gray-800 rounded-lg hover:bg-gray-300 transition-colors"
          >
            Done
          </button>
        </div>
      </div>
    </div>
  );

  const getRankColor = (rank) => {
    if (rank === 1) return 'text-yellow-600 bg-yellow-50';
    if (rank <= 3) return 'text-orange-600 bg-orange-50';
    if (rank <= 6) return 'text-green-600 bg-green-50';
    if (rank <= 10) return 'text-blue-600 bg-blue-50';
    return 'text-gray-600 bg-gray-50';
  };

  const getFormColor = (form) => {
    if (form >= 5) return 'text-green-700 bg-green-100';
    if (form >= 2) return 'text-green-600 bg-green-50';
    if (form >= -2) return 'text-gray-600 bg-gray-50';
    if (form >= -5) return 'text-red-600 bg-red-50';
    return 'text-red-700 bg-red-100';
  };

  return (
    <div className="max-w-7xl mx-auto p-4 bg-gray-50 min-h-screen">
      <div className="bg-white rounded-xl shadow-lg overflow-hidden">
        <div className="bg-gradient-to-r from-blue-600 to-purple-600 text-white p-6">
          <h1 className="text-3xl font-bold flex items-center gap-3">
            <Trophy size={32} />
            OG JITS S6 (2025) Enhanced Power Rankings
          </h1>
          <p className="text-blue-100 mt-2">Advanced analytics with strength of schedule and opponent quality</p>
        </div>

        <div className="border-b border-gray-200">
          <nav className="flex">
            {[
              { id: 'rankings', label: 'Power Rankings', icon: Trophy },
              { id: 'analytics', label: 'Analytics', icon: TrendingUp },
              { id: 'schedule', label: 'Schedule Analysis', icon: Calendar }
            ].map(tab => (
              <button
                key={tab.id}
                onClick={() => setActiveTab(tab.id)}
                className={`flex items-center gap-2 px-6 py-4 font-medium transition-colors ${
                  activeTab === tab.id 
                    ? 'text-blue-600 border-b-2 border-blue-600 bg-blue-50' 
                    : 'text-gray-600 hover:text-gray-900 hover:bg-gray-50'
                }`}
              >
                <tab.icon size={18} />
                {tab.label}
              </button>
            ))}
          </nav>
        </div>

        <div className="p-6">
          {activeTab === 'rankings' && (
            <div className="space-y-6">
              <div className="flex justify-between items-center">
                <h2 className="text-xl font-bold">Current Power Rankings</h2>
                <div className="text-sm text-gray-600">
                  Updated: {new Date().toLocaleDateString()}
                </div>
              </div>

              <div className="overflow-x-auto">
                <table className="w-full">
                  <thead>
                    <tr className="bg-gray-50">
                      <th className="text-left p-4 font-semibold">Rank</th>
                      <th className="text-left p-4 font-semibold">Team</th>
                      <th className="text-center p-4 font-semibold">Record</th>
                      <th className="text-center p-4 font-semibold">Win%</th>
                      <th className="text-center p-4 font-semibold">Point Diff</th>
                      <th className="text-center p-4 font-semibold">SOS</th>
                      <th className="text-center p-4 font-semibold">Form</th>
                      <th className="text-center p-4 font-semibold">Power Rating</th>
                      <th className="text-center p-4 font-semibold">Actions</th>
                    </tr>
                  </thead>
                  <tbody>
                    {enhancedTeams.map((team, index) => (
                      <tr key={team.id} className="border-b border-gray-100 hover:bg-gray-50">
                        <td className="p-4">
                          <span className={`inline-flex items-center justify-center w-8 h-8 rounded-full font-bold text-sm ${getRankColor(index + 1)}`}>
                            {index + 1}
                          </span>
                        </td>
                        <td className="p-4">
                          <div className="font-semibold text-gray-900">{team.name}</div>
                          <div className="text-sm text-gray-600">
                            QW: {team.qualityWins} | BL: {team.badLosses}
                          </div>
                        </td>
                        <td className="p-4 text-center font-mono">
                          <div className="font-semibold">{team.wins}-{team.losses}</div>
                          <div className="text-sm text-gray-600">({team.gamesPlayed} games)</div>
                        </td>
                        <td className="p-4 text-center font-mono">
                          {(team.winPct * 100).toFixed(1)}%
                        </td>
                        <td className="p-4 text-center font-mono">
                          <div className={team.pointDiff >= 0 ? 'text-green-600' : 'text-red-600'}>
                            {team.pointDiff >= 0 ? '+' : ''}{team.pointDiff}
                          </div>
                          <div className="text-xs text-gray-500">
                            {team.ppg.toFixed(1)} - {team.papg.toFixed(1)}
                          </div>
                        </td>
                        <td className="p-4 text-center font-mono">
                          <div className={team.sos >= 0 ? 'text-orange-600' : 'text-green-600'}>
                            {team.sos >= 0 ? '+' : ''}{(team.sos * 100).toFixed(1)}%
                          </div>
                          <div className="text-xs text-gray-500">
                            vs {(team.opponentWinPct * 100).toFixed(1)}%
                          </div>
                        </td>
                        <td className="p-4 text-center">
                          <span className={`inline-flex items-center px-2 py-1 rounded-full text-xs font-medium ${getFormColor(team.recentForm)}`}>
                            {team.recentForm >= 0 ? '+' : ''}{team.recentForm.toFixed(1)}
                          </span>
                        </td>
                        <td className="p-4 text-center font-mono font-bold text-lg">
                          {team.powerRating.toFixed(1)}
                        </td>
                        <td className="p-4 text-center">
                          <button
                            onClick={() => setEditingTeam(team)}
                            className="p-2 text-gray-600 hover:text-blue-600 hover:bg-blue-50 rounded-lg transition-colors"
                          >
                            <Edit3 size={16} />
                          </button>
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            </div>
          )}

          {activeTab === 'analytics' && (
            <div className="space-y-6">
              <h2 className="text-xl font-bold">League Analytics</h2>
              
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
                <div className="bg-blue-50 p-4 rounded-lg">
                  <div className="text-blue-600 text-sm font-medium">Average Win %</div>
                  <div className="text-2xl font-bold text-blue-900">
                    {((enhancedTeams.reduce((sum, team) => sum + team.winPct, 0) / enhancedTeams.length) * 100).toFixed(1)}%
                  </div>
                </div>
                
                <div className="bg-green-50 p-4 rounded-lg">
                  <div className="text-green-600 text-sm font-medium">League PPG</div>
                  <div className="text-2xl font-bold text-green-900">
                    {(enhancedTeams.reduce((sum, team) => sum + team.ppg, 0) / enhancedTeams.length).toFixed(1)}
                  </div>
                </div>
                
                <div className="bg-orange-50 p-4 rounded-lg">
                  <div className="text-orange-600 text-sm font-medium">Avg SOS</div>
                  <div className="text-2xl font-bold text-orange-900">
                    {((enhancedTeams.reduce((sum, team) => sum + team.sos, 0) / enhancedTeams.length) * 100).toFixed(1)}%
                  </div>
                </div>
                
                <div className="bg-purple-50 p-4 rounded-lg">
                  <div className="text-purple-600 text-sm font-medium">Power Range</div>
                  <div className="text-2xl font-bold text-purple-900">
                    {(enhancedTeams[0].powerRating - enhancedTeams[enhancedTeams.length - 1].powerRating).toFixed(1)}
                  </div>
                </div>
              </div>

              <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
                <div className="bg-white border rounded-lg p-4">
                  <h3 className="font-bold mb-3">Top Performers</h3>
                  <div className="space-y-2">
                    {enhancedTeams.slice(0, 5).map((team, index) => (
                      <div key={team.id} className="flex justify-between items-center p-2 rounded bg-gray-50">
                        <span className="font-medium">{index + 1}. {team.name}</span>
                        <span className="font-mono font-bold">{team.powerRating.toFixed(1)}</span>
                      </div>
                    ))}
                  </div>
                </div>

                <div className="bg-white border rounded-lg p-4">
                  <h3 className="font-bold mb-3">Strongest Schedules</h3>
                  <div className="space-y-2">
                    {enhancedTeams
                      .sort((a, b) => b.sos - a.sos)
                      .slice(0, 5)
                      .map((team, index) => (
                        <div key={team.id} className="flex justify-between items-center p-2 rounded bg-gray-50">
                          <span className="font-medium">{team.name}</span>
                          <span className="font-mono text-orange-600">
                            {team.sos >= 0 ? '+' : ''}{(team.sos * 100).toFixed(1)}%
                          </span>
                        </div>
                      ))}
                  </div>
                </div>
              </div>
            </div>
          )}

          {activeTab === 'schedule' && (
            <div className="space-y-6">
              <h2 className="text-xl font-bold">Schedule Analysis</h2>
              
              <div className="overflow-x-auto">
                <table className="w-full">
                  <thead>
                    <tr className="bg-gray-50">
                      <th className="text-left p-4 font-semibold">Team</th>
                      <th className="text-center p-4 font-semibold">Opponent Win%</th>
                      <th className="text-center p-4 font-semibold">SOS Rating</th>
                      <th className="text-center p-4 font-semibold">Quality Wins</th>
                      <th className="text-center p-4 font-semibold">Bad Losses</th>
                      <th className="text-center p-4 font-semibold">Schedule Grade</th>
                    </tr>
                  </thead>
                  <tbody>
                    {enhancedTeams
                      .sort((a, b) => b.sos - a.sos)
                      .map((team) => {
                        const grade = team.sos > 0.05 ? 'A+' : 
                                     team.sos > 0.02 ? 'A' :
                                     team.sos > -0.02 ? 'B' :
                                     team.sos > -0.05 ? 'C' : 'D';
                        const gradeColor = grade.startsWith('A') ? 'text-red-600 bg-red-50' :
                                          grade === 'B' ? 'text-orange-600 bg-orange-50' :
                                          'text-green-600 bg-green-50';
                        
                        return (
                          <tr key={team.id} className="border-b border-gray-100 hover:bg-gray-50">
                            <td className="p-4 font-semibold">{team.name}</td>
                            <td className="p-4 text-center font-mono">
                              {(team.opponentWinPct * 100).toFixed(1)}%
                            </td>
                            <td className="p-4 text-center font-mono">
                              <span className={team.sos >= 0 ? 'text-orange-600' : 'text-green-600'}>
                                {team.sos >= 0 ? '+' : ''}{(team.sos * 100).toFixed(1)}%
                              </span>
                            </td>
                            <td className="p-4 text-center font-mono text-green-600">
                              {team.qualityWins}
                            </td>
                            <td className="p-4 text-center font-mono text-red-600">
                              {team.badLosses}
                            </td>
                            <td className="p-4 text-center">
                              <span className={`inline-flex items-center px-2 py-1 rounded-full text-sm font-bold ${gradeColor}`}>
                                {grade}
                              </span>
                            </td>
                          </tr>
                        );
                      })}
                  </tbody>
                </table>
              </div>

              <div className="bg-blue-50 p-4 rounded-lg">
                <h3 className="font-bold text-blue-900 mb-2">Schedule Difficulty Legend</h3>
                <div className="grid grid-cols-2 md:grid-cols-5 gap-2 text-sm">
                  <div className="text-red-700"><strong>A+:</strong> Brutal (+5%)</div>
                  <div className="text-red-600"><strong>A:</strong> Very Hard (+2%)</div>
                  <div className="text-orange-600"><strong>B:</strong> Average (±2%)</div>
                  <div className="text-green-600"><strong>C:</strong> Easy (-2%)</div>
                  <div className="text-green-700"><strong>D:</strong> Very Easy (-5%)</div>
                </div>
              </div>
            </div>
          )}
        </div>
      </div>

      {editingTeam && (
        <TeamEditor 
          team={editingTeam} 
          onClose={() => setEditingTeam(null)} 
        />
      )}
    </div>
  );
};

export default PowerRankingsSystem;
